\begin{frame}{Lý do chọn}
    \begin{itemize}
        \item Mã nguồn mở.
        \item Phương pháp đơn giản, nhưng hiệu quả.
        \item Yêu cầu về thực nghiệm (dữ liệu, chi phí) nằm trong khả năng của nhóm.
        \item \textit{Chú ý về dữ liệu:}
              \begin{itemize}
                  \item Nghiên cứu gốc thực nghiệm với 4 tập dữ liệu: CASIA-B, OU-MVLP, GREW, Gait3D.
                  \item Duy nhất tập CASIA-B có thể truy cập và sử dụng dễ dàng, đồng thời kích thước nhỏ. Các tập khác đều cần quyền truy cập từ tác giả (phải kèm theo đại diện của Trưởng Khoa/Trưởng Nhóm nghiên cứu), thời gian xác nhận dài. Kích thước rất lớn, không khả thi.
              \end{itemize}
    \end{itemize}
\end{frame}
