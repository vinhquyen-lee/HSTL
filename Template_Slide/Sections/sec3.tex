\begin{frame}{Conv3D và P3D}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Conv3D (Tiêu chuẩn)}
        $$ W_{3D} \in \mathbb{R}^{k \times k \times k} $$
        \begin{equation}
            y = W_{3D} \ast x
        \end{equation}
        Chi phí: $\mathcal{O}(k^3)$
        \column{0.5\textwidth}
        \textbf{P3D-A (Đề xuất)}
        $$ W_S \in \mathbb{R}^{1 \times k \times k}, \quad W_T \in \mathbb{R}^{k \times 1 \times 1} $$
        \begin{equation}
            x_{t+1} = x_t + W_T \ast (W_S \ast x_t)
        \end{equation}
        Chi phí: $\mathcal{O}(k^2+k)$
    \end{columns}
\end{frame}

\begin{frame}{Triplet Loss}
    \begin{equation}
        \max(0, d(x^a, x^p) - d(x^a, x^n) + m)
    \end{equation}
    \textbf{Trong đó:}
    \begin{itemize}
        \item $d(x, y) = ||f(x)-f(y)||_2^2$: Khoảng cách Euclidean.
        \item $x^a, x^p, x^n$: Điểm neo, điểm dương, điểm âm.
        \item $m$: biên.
    \end{itemize}
    % \textbf{Gradient (Hằng số):}
    % $$ \frac{\partial \mathcal{L}}{\partial d_p} = 1, \quad \frac{\partial \mathcal{L}}{\partial d_n} = -1 \quad (\text{khi loss} > 0) $$
\end{frame}

% --- CIRCLE LOSS ---
\begin{frame}{Circle Loss}
    \begin{equation}
        \mathcal{L}_{circle} = \log\left[ 1 + \sum_{j} e^{\gamma \alpha_n^j (s_n^j - \Delta_n)} \cdot \sum_{i} e^{-\gamma \alpha_p^i (s_p^i - \Delta_p)} \right]
    \end{equation}
    \textbf{Trong đó:}
    \begin{itemize}
        \item $s_p = \cos(x^a, x^p), \quad s_n = \cos(x^a, x^n)$.
    \end{itemize}
\end{frame}

\begin{frame}{Triplet Loss và Circle Loss}
    \begin{table}
        \centering
        \resizebox{\textwidth}{!}{
            \begin{tabular}{lll}
                \toprule
                \textbf{Đặc điểm} & \textbf{Triplet Loss}            & \textbf{Circle Loss}               \\
                \midrule
                Đơn vị đo         & Khoảng cách Euclidean            & Độ tương đồng Cosine               \\
                Mục tiêu          & Tương đối ($d_n - d_p > \alpha$) & Tuyệt đối ($s_p \to 1, s_n \to 0$) \\
                Gradient          & Hằng số                          & Tự điều chỉnh (Lớn cho mẫu khó)    \\
                % Mining            & Cần Hard Mining                  & Tự động (Self-paced)               \\
                \bottomrule
            \end{tabular}
        }
        \caption{So sánh Triplet Loss và Circle Loss.}
    \end{table}
\end{frame}
\begin{frame}{Hàm mất mát tổng hợp}
    \textbf{Phiên bản gốc:}
    \begin{equation}
        \mathcal{L} = \lambda_{triplet} \mathcal{L}_{triplet} + \lambda_{ce} \mathcal{L}_{ce}
    \end{equation}
    \textbf{Phiên bản đề xuất:}
    \begin{equation}
        \mathcal{L} = \mathcal{L}_{circle}
    \end{equation}
\end{frame}
