\begin{frame}{Phân tích lỗi thực nghiệm Circle Loss}
    % \begin{alertblock}{Sai lầm trong cài đặt thực nghiệm}
    \begin{itemize}
        \item \textbf{Kỳ vọng:} Thay thế tổ hợp (Triplet + CE) bằng Circle Loss.
        \item \textbf{Thực tế đã làm:} Chỉ thay thế Triplet Loss bằng Circle Loss, vẫn giữ lại CE Loss.
        \item \textbf{Hậu quả:} Sai nguyên lý hoạt động, dẫn đến hiệu năng giảm sút.
    \end{itemize}
