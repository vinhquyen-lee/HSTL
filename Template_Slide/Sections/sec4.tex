\subsection*{Huấn luyện}

% - tập dữ liệu ImageNet100k dùng để huấn luyện mô hình tổng quát, với mong muốn tô màu trên nhiều miền ảnh nhất có thể.
% - ba tập dữ liệu còn lại dùng để huấn luyện các mô hình tô màu trên miền ảnh chuyên biệt, bao gồm thời trang, nội thất, và khuôn mặt

\begin{frame}{Tập dữ liệu và Giao thức đánh giá}
    \begin{block}{Tập dữ liệu CASIA-B}
        Tập dữ liệu chuẩn phổ biến nhất cho bài toán nhận diện dáng đi trong phòng Lab.
        \begin{itemize}
            \item \textbf{Số lượng:} 124 chủ thể (Subjects).
            \item \textbf{Góc nhìn (Viewpoints):} 11 góc ($0^{\circ}$ đến $180^{\circ}$, bước nhảy $18^{\circ}$).
            \item \textbf{Điều kiện đi bộ (3 loại):}
            \begin{itemize}
                \item \textbf{NM (Normal):} Đi bình thường (6 video/người).
                \item \textbf{BG (Bag):} Mang túi xách (2 video/người).
                \item \textbf{CL (Coat):} Mặc áo khoác dày (2 video/người).
            \end{itemize}
        \end{itemize}
    \end{block}
\end{frame}
\begin{frame}{Tập dữ liệu và Giao thức đánh giá}
    \begin{alertblock}{Giao thức đánh giá (Evaluation Protocol)}
        \begin{itemize}
            \item \textbf{Phân chia:} Train (74 chủ thể đầu) / Test (50 chủ thể còn lại).
            \item \textbf{Trong tập Test:}
            \begin{itemize}
                \item \textit{Gallery (Tham chiếu):} 4 video NM đầu tiên (\#01--\#04).
                \item \textit{Probe (Truy vấn):} Các video còn lại (NM \#05-06, BG, CL).
            \end{itemize}
            \item \textbf{Độ đo:} Rank-1 Accuracy (loại trừ các cặp cùng góc nhìn).
        \end{itemize}
    \end{alertblock}
\end{frame}
% các tập dữ liệu sẽ được tiền xử lý qua 3 bước:
% - lọc các ảnh tương đối xám, chỉ giữa lại các ảnh  màu sắc ổn định để huấn luyện
% - điều chỉnh về cùng 1 độ phân giải
% - tạo sinh câu mô tả bằng mô hình BLIP để làm điều kiện cho quá trình huấn luyệnluyện

\begin{frame}{Cài đặt thực nghiệm}
    \begin{block}{Quy trình tiền xử lý (Data Preprocessing)}
        Dữ liệu đầu vào là các chuỗi ảnh dáng đi (Silhouette) được xử lý như sau:
        
        \vspace{0.3cm} % Tạo khoảng cách cho thoáng
        \begin{itemize}
            \setlength\itemsep{1em} % Tăng khoảng cách giữa các gạch đầu dòng
            \item \textbf{Đầu vào (Input):} 
            Chuỗi ảnh dáng đi đã được căn chỉnh (aligned) và chuẩn hóa.
            
            \item \textbf{Kích thước ảnh:} 
            Mỗi khung hình được cắt và thay đổi kích thước về \alert{$64 \times 44$ pixel}.
            
            \item \textbf{Chiến lược lấy mẫu (Sampling Strategy):}
            \begin{itemize}
                \vspace{0.2cm}
                \item \textbf{Huấn luyện (Train):} Chọn ngẫu nhiên \textbf{30 khung hình} từ mỗi chuỗi video để đảm bảo tính đa dạng dữ liệu.
                \item \textbf{Kiểm thử (Test):} Sử dụng \textbf{toàn bộ} các khung hình trong chuỗi để trích xuất đặc trưng đầy đủ nhất.
            \end{itemize}
        \end{itemize}
    \end{block}
\end{frame}

% --- SLIDE 2: CẤU HÌNH MÔ HÌNH ---
\begin{frame}{Cài đặt thực nghiệm}
    \begin{exampleblock}{Cấu hình kiến trúc HSTL-P3D}
        Mô hình được thiết kế chuyên biệt cho CASIA-B với các thông số chi tiết:
        
        \vspace{0.3cm}
        \begin{itemize}
            \setlength\itemsep{1em}
            \item \textbf{Phân hoạch không gian:} 
            Cơ thể được chia thành \textbf{$K=8$ phần} theo chiều dọc (Horizontal Mapping).
            
            \item \textbf{Kiến trúc phân cấp (Hierarchical Structure):}
            Hệ thống gồm 4 tầng xử lý chính:
            \begin{enumerate}
                \setlength\itemsep{0.5em} % Tăng khoảng cách giữa các số 1,2,3,4
                \item \textbf{Level 1:} ARME Module (Output Channels: 32).
                \item \textbf{Level 2:} ARME Module (Output Channels: 64) + Nén thời gian (FTA).
                \item \textbf{Level 3:} ARME Module (Output Channels: 128) + Gom nhóm bộ phận.
                \item \textbf{Level 4:} Phân hoạch đều (Uniform Partition) để tạo vector đặc trưng.
            \end{enumerate}
        \end{itemize}
    \end{exampleblock}
\end{frame}

\subsection*{Đánh giá}
% - Độ đo FID và Colorfulness được dùng để đánh giá tất cả phương thức tô màu của mô hình
% - Điểm số CLIP được dùng riêng cho đánh giá phương thức tô màu dựa trên câu mô tả

\begin{frame}{Độ đo}
Có ba độ đo phổ biến trong các nghiên cứu tô màu ảnh độ xám, gồm:
\begin{itemize}
    \item \textbf{Độ đo Fréchet Inception Distance (FID)} - kiểm tra độ chân thật của ảnh được tạo ra so với ảnh gốc. Ảnh sinh ra càng giống ảnh gốc thì FID càng thấp.
    \item \textbf{Độ đo Colorfulness} - đánh giá độ sặc sỡ của ảnh được tô màu. Ảnh càng sặc sỡ thì Colorfulness càng cao.
    \item \textbf{Điểm số CLIP} - đánh giá độ tương đồng giữa ảnh kết quả với câu mô tả trong không gian tiềm ẩn của CLIP, được dùng cho phương thức tô màu bán tự động.
    
    % \[
    %     \text{CLIP score} = \dfrac{\mathrm{I} \cdot \mathrm{T}}{||\mathrm{I}||\cdot ||\mathrm{T}||}
    % \]
    % trong đó, $\mathrm{I} = \epsilon_i(i)$ và $\mathrm{T} = \epsilon_t(t)$, lần lượt là vec-tơ mã hóa của ảnh và văn bản của hai bộ mã hóa tương ứng.
\end{itemize}
\end{frame}
